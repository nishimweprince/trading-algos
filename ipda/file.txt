// ── Supertrend ────────────────────────────────────────────────────────────────
supertrend_f(_src, factor, atrLen) =>
    atrat         = ta.atr(atrLen)
    upperBand     = _src + factor * atrat
    lowerBand     = _src - factor * atrat
    prevLowerBand = nz(lowerBand[1])
    prevUpperBand = nz(upperBand[1])
    lowerBand    := lowerBand > prevLowerBand or close[1] < prevLowerBand ? lowerBand : prevLowerBand
    upperBand    := upperBand < prevUpperBand or close[1] > prevUpperBand ? upperBand : prevUpperBand
    int   direction  = na
    float superTrend = na
    prevSuperTrend   = superTrend[1]
    if na(atrat[1])
        direction := 1
    else if prevSuperTrend == prevUpperBand
        direction := close > upperBand ? -1 : 1
    else
        direction := close < lowerBand ? 1 : -1
    superTrend := direction == -1 ? lowerBand : upperBand
    [superTrend, direction]

// ── HTF Security ─────────────────────────────────────────────────────────────
tf_f(_res, _exp, gaps_on) =>
    gaps_on == 0    ? request.security(syminfo.tickerid, _res, _exp) :
     gaps_on == true ? request.security(syminfo.tickerid, _res, _exp, barmerge.gaps_on,  barmerge.lookahead_off) :
                       request.security(syminfo.tickerid, _res, _exp, barmerge.gaps_off, barmerge.lookahead_off)

// ── Range Filter helpers ──────────────────────────────────────────────────────
Cond_EMA(x, cond, n) =>
    var val     = array.new_float(0)
    var ema_val = array.new_float(1)
    if cond
        array.push(val, x)
        if array.size(val) > 1
            array.remove(val, 0)
        if na(array.get(ema_val, 0))
            array.fill(ema_val, array.get(val, 0))
        array.set(ema_val, 0, (array.get(val, 0) - array.get(ema_val, 0)) * (2.0 / (n + 1)) + array.get(ema_val, 0))
    array.get(ema_val, 0)

Cond_SMA(x, cond, n) =>
    var vals = array.new_float(0)
    if cond
        array.push(vals, x)
        if array.size(vals) > n
            array.remove(vals, 0)
    array.avg(vals)

Stdev(x, n) =>
    math.sqrt(Cond_SMA(math.pow(x, 2), 1, n) - math.pow(Cond_SMA(x, 1, n), 2))

rng_size_f(x, scale, qty, n) =>
    ATR_rf = Cond_EMA(ta.tr(true), 1, n)
    AC_rf  = Cond_EMA(math.abs(x - x[1]), 1, n)
    SD_rf  = Stdev(x, n)
    scale == "Pips"               ? qty * 0.0001             :
     scale == "Points"            ? qty * syminfo.pointvalue  :
     scale == "% of Price"        ? close * qty / 100         :
     scale == "ATR"               ? qty * ATR_rf              :
     scale == "Average Change"    ? qty * AC_rf               :
     scale == "Standard Deviation"? qty * SD_rf               :
     scale == "Ticks"             ? qty * syminfo.mintick     : qty

rng_filt_f(h, l, rng_, n, ftype, smooth, sn, av_rf, av_n) =>
    rng_smooth = Cond_EMA(rng_, 1, sn)
    r          = smooth ? rng_smooth : rng_
    var rfilt  = array.new_float(2, (h + l) / 2)
    array.set(rfilt, 1, array.get(rfilt, 0))
    if ftype == "Type 1"
        if h - r > array.get(rfilt, 1)
            array.set(rfilt, 0, h - r)
        if l + r < array.get(rfilt, 1)
            array.set(rfilt, 0, l + r)
    if ftype == "Type 2"
        if h >= array.get(rfilt, 1) + r
            array.set(rfilt, 0, array.get(rfilt, 1) + math.floor(math.abs(h - array.get(rfilt, 1)) / r) * r)
        if l <= array.get(rfilt, 1) - r
            array.set(rfilt, 0, array.get(rfilt, 1) - math.floor(math.abs(l - array.get(rfilt, 1)) / r) * r)
    rng_filt1 = array.get(rfilt, 0)
    hi_band1  = rng_filt1 + r
    lo_band1  = rng_filt1 - r
    rng_filt2 = Cond_EMA(rng_filt1, rng_filt1 != rng_filt1[1], av_n)
    hi_band2  = Cond_EMA(hi_band1,  rng_filt1 != rng_filt1[1], av_n)
    lo_band2  = Cond_EMA(lo_band1,  rng_filt1 != rng_filt1[1], av_n)
    rng_out   = av_rf ? rng_filt2 : rng_filt1
    hi_out    = av_rf ? hi_band2  : hi_band1
    lo_out    = av_rf ? lo_band2  : lo_band1
    [hi_out, lo_out, rng_out]

// ── SuperIchi helper ──────────────────────────────────────────────────────────
avg_ichi(srcc, length, mult) =>
    atr_i   = ta.atr(length) * mult
    up_i    = hl2 + atr_i
    dn_i    = hl2 - atr_i
    float upper_i = 0.0
    float lower_i = 0.0
    upper_i := srcc[1] < upper_i[1] ? math.min(up_i, upper_i[1]) : up_i
    lower_i := srcc[1] > lower_i[1] ? math.max(dn_i, lower_i[1]) : dn_i
    int   os_i  = 0
    float max_i = 0.0
    float min_i = 0.0
    os_i  := srcc > upper_i ? 1 : srcc < lower_i ? 0 : os_i[1]
    spt_i  = os_i == 1 ? lower_i : upper_i
    max_i := ta.cross(srcc, spt_i) ? math.max(srcc, max_i[1]) : os_i == 1 ? math.max(srcc, max_i[1]) : spt_i
    min_i := ta.cross(srcc, spt_i) ? math.min(srcc, min_i[1]) : os_i == 0 ? math.min(srcc, min_i[1]) : spt_i
    math.avg(max_i, min_i)

// ── TBO Bollinger ─────────────────────────────────────────────────────────────
bb_tbo(src, len, mult) =>
    float basis_t = ta.ema(src, len)
    float dev_t   = mult * ta.stdev(src, len)
    [basis_t, basis_t + dev_t, basis_t - dev_t]

// ── Smart Trail Wilder MA ─────────────────────────────────────────────────────
Wild_ma(_src, _malength) =>
    float _wild = 0.0
    _wild := nz(_wild[1]) + (_src - nz(_wild[1])) / _malength
    _wild

// ── TP Points (Lele) ──────────────────────────────────────────────────────────
lele(qual, len) =>
    float bindex = 0.0
    float sindex = 0.0
    bindex := nz(bindex[1], 0)
    sindex := nz(sindex[1], 0)
    int ret = 0
    if close > close[4]
        bindex += 1
    if close < close[4]
        sindex += 1
    if bindex > qual and close < open and high >= ta.highest(high, len)
        bindex := 0
        ret    := -1
    if sindex > qual and close > open and low <= ta.lowest(low, len)
        sindex := 0
        ret    := 1
    ret

// ── KAMA (Reversal Band) ──────────────────────────────────────────────────────
kama_f(ssrc, llen) =>
    float kv = 0.0
    sum_1     = math.sum(math.abs(ssrc - ssrc[1]), llen)
    sum_2     = math.sum(math.abs(ssrc - ssrc[1]), llen)
    kv       := nz(kv[1]) + math.pow((sum_1 != 0 ? math.abs(ssrc - ssrc[llen]) / sum_2 : 0) * (0.288 - 0.0666) + 0.0666, 2) * (ssrc - nz(kv[1]))
    kv

// ── SR percent width ──────────────────────────────────────────────────────────
percWidth(len, perc) => (ta.highest(len) - ta.lowest(len)) * perc / 100

// ── IFVG drawing helpers ──────────────────────────────────────────────────────

ifvg_c_top() => math.max(open, close)
ifvg_c_bot() => math.min(open, close)

// ── IFVG: move FVG to inversion array when price enters it ───────────────────
// NOTE: we pass drawing-arrays in so we can clean up IFVG-owned objects only
ifvg_fvg_manage(_ary, _inv_ary) =>
    ifvg_buf = 100
    if _ary.size() >= ifvg_buf
        _ary.shift()
    if _ary.size() > 0
        for i = _ary.size() - 1 to 0
            value = _ary.get(i)
            _dir  = value.dir
            if _dir == 1  and (ifvg_c_bot() < value.bot)
                value.x_val := time
                _inv_ary.push(_ary.remove(i))
            if _dir == -1 and (ifvg_c_top() > value.top)
                value.x_val := time
                _inv_ary.push(_ary.remove(i))

// ── IFVG: manage inversion entries and fire signals ──────────────────────────
ifvg_inv_manage(_ary) =>
    ifvg_buf = 100
    wt       = signal_pref == "Wick"
    fire     = false
    if _ary.size() >= ifvg_buf
        _ary.shift()
    if _ary.size() > 0
        for i = _ary.size() - 1 to 0
            value  = _ary.get(i)
            bx_top = value.top
            bx_bot = value.bot
            _dir   = value.dir
            st     = value.state
            if st == 0 and _dir ==  1
                value.state := 1
                value.dir   := -1
            if st == 0 and _dir == -1
                value.state := 1
                value.dir   :=  1
            if st >= 1
                value.right := time
            if _dir == -1 and st == 1 and close < bx_bot and (wt ? high : close[1]) >= bx_bot and (wt ? high : close[1]) < bx_top
                value.labs.push(ifvg_lab.new(time, bx_top, -1))
                fire := true
            if _dir ==  1 and st == 1 and close > bx_top and (wt ? low  : close[1]) <= bx_top and (wt ? low  : close[1]) > bx_bot
                value.labs.push(ifvg_lab.new(time, bx_bot,  1))
                fire := true
            if st >= 1 and ((_dir == -1 and ifvg_c_top() > bx_top) or (_dir == 1 and ifvg_c_bot() < bx_bot))
                _ary.remove(i)
    fire

    array.push(ifvg_bull_fvg_ary, ifvg_fvg.new(time[1], low, time, high[2], math.avg(low, high[2]),  1, 0, array.new<ifvg_lab>(), na))

if fvg_down_i and math.abs(low[2] - high) > ifvg_atr
    array.push(ifvg_bear_fvg_ary, ifvg_fvg.new(time[1], low[2], time, high, math.avg(high, low[2]), -1, 0, array.new<ifvg_lab>(), na))

// Clean up previous IFVG drawings only (no nuke of all objects)
for bx in ifvg_boxes
    box.delete(bx)
for ln in ifvg_lines
    line.delete(ln)
for lb in ifvg_labels
    label.delete(lb)

array.clear(ifvg_boxes)
array.clear(ifvg_lines)
array.clear(ifvg_labels)

ifvg_fvg_manage(ifvg_bull_fvg_ary, ifvg_bull_inv_ary)
ifvg_fvg_manage(ifvg_bear_fvg_ary, ifvg_bear_inv_ary)

bear_ifvg_signal = ifvg_inv_manage(ifvg_bull_inv_ary)
bull_ifvg_signal = ifvg_inv_manage(ifvg_bear_inv_ary)

if barstate.islast
    ifvg_send_it(ifvg_bull_inv_ary, ifvg_boxes, ifvg_lines, ifvg_labels)
    ifvg_send_it(ifvg_bear_inv_ary, ifvg_boxes, ifvg_lines, ifvg_labels)

if bull_ifvg_signal and alert_ifvg_bull
    alert("🟢 IFVG Bullish Signal — " + syminfo.ticker + " @ " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)
if bear_ifvg_signal and alert_ifvg_bear
    alert("🔴 IFVG Bearish Signal — " + syminfo.ticker + " @ " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)

alertcondition(bull_ifvg_signal, "IFVG Bullish Signal", "IFVG Bullish Signal")
alertcondition(bear_ifvg_signal, "IFVG Bearish Signal", "IFVG Bearish Signal")
